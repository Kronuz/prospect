cmake_minimum_required(VERSION 3.14)
project(strata LANGUAGES C CXX)

include(FetchContent)

# Building blocks (Kronuz libraries). Point FETCHCONTENT_SOURCE_DIR_<NAME> at local
# checkouts to develop them together.
set(TASK_ASIO OFF CACHE BOOL "" FORCE)
# Each building block is pinned to the commit its Phase 1 gate numbers were measured at.
FetchContent_Declare(docset GIT_REPOSITORY https://github.com/Kronuz/docset.git GIT_TAG 4e4d73fda4b53583122e8b1a976b96c496cee3b2)
FetchContent_Declare(bitpack GIT_REPOSITORY https://github.com/Kronuz/bitpack.git GIT_TAG 6d14df141e1dc9c8f5e709f11a55f48493744370)
FetchContent_Declare(fst GIT_REPOSITORY https://github.com/Kronuz/fst.git GIT_TAG a8d8478e8f92661ecf7e82eb85ca827155da2cad)
FetchContent_Declare(task GIT_REPOSITORY https://github.com/Kronuz/task.git GIT_TAG c3c30e49212769e1bbf424517003150de3f20682)
FetchContent_Declare(hashes GIT_REPOSITORY https://github.com/Kronuz/hashes.git GIT_TAG abff6fa5ac0c19c0cff5509a437700d8f77be41e)
FetchContent_Declare(termdict GIT_REPOSITORY https://github.com/Kronuz/termdict.git GIT_TAG 4100fda23468b5502df4bc4aac0bf5f3c0e7a4be)
FetchContent_Declare(varint GIT_REPOSITORY https://github.com/Kronuz/varint.git GIT_TAG de3d05dbec86cd4c147b3177947f9c3b58312675)
FetchContent_Declare(residency GIT_REPOSITORY https://github.com/Kronuz/residency.git GIT_TAG e52bd7bcb987a7f07993f1dc73b1760453067db2)
FetchContent_Declare(unicode GIT_REPOSITORY https://github.com/Kronuz/unicode.git GIT_TAG 5c745dd6a44de27123ce7ef2f80c1049f8eec04b)
FetchContent_Declare(snowball GIT_REPOSITORY https://github.com/Kronuz/snowball.git GIT_TAG main)
FetchContent_Declare(analysis GIT_REPOSITORY https://github.com/Kronuz/analysis.git GIT_TAG main)
FetchContent_MakeAvailable(docset bitpack fst task hashes residency varint termdict unicode snowball analysis)

add_library(strata INTERFACE)
target_include_directories(strata INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(strata INTERFACE cxx_std_20)
target_link_libraries(strata INTERFACE docset::docset bitpack::bitpack fst::fst task::task hashes::hashes residency::residency varint::varint termdict::termdict analysis::analysis)
add_library(strata::strata ALIAS strata)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  enable_testing()
  add_executable(concepts_test test/concepts_test.cc)
  target_link_libraries(concepts_test PRIVATE strata)
  add_test(NAME concepts COMMAND concepts_test)
  add_executable(segment_test test/segment_test.cc)
  target_link_libraries(segment_test PRIVATE strata)
  target_compile_definitions(segment_test PRIVATE STRATA_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/test/fixtures")
  add_test(NAME segment COMMAND segment_test)
  add_executable(search_test test/search_test.cc)
  target_link_libraries(search_test PRIVATE strata)
  add_test(NAME search COMMAND search_test)
  add_executable(parity tools/parity.cc)
  target_link_libraries(parity PRIVATE strata)
  add_executable(strata_engine tools/strata_engine.cc)
  target_link_libraries(strata_engine PRIVATE strata)
endif()
