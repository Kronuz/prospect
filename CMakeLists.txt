cmake_minimum_required(VERSION 3.18)
project(strata LANGUAGES C CXX)

include(FetchContent)

# Building blocks (Kronuz libraries). Point FETCHCONTENT_SOURCE_DIR_<NAME> at local
# checkouts to develop them together.
set(TASK_ASIO OFF CACHE BOOL "" FORCE)
# Each building block is pinned to the commit its Phase 1 gate numbers were measured at.
FetchContent_Declare(docset GIT_REPOSITORY https://github.com/Kronuz/docset.git GIT_TAG 4e4d73fda4b53583122e8b1a976b96c496cee3b2)
FetchContent_Declare(bitpack GIT_REPOSITORY https://github.com/Kronuz/bitpack.git GIT_TAG 0ce26911f8e8ab0d59f7de1930efcec83f9f7d12)
FetchContent_Declare(fst GIT_REPOSITORY https://github.com/Kronuz/fst.git GIT_TAG a8d8478e8f92661ecf7e82eb85ca827155da2cad)
FetchContent_Declare(task GIT_REPOSITORY https://github.com/Kronuz/task.git GIT_TAG c3c30e49212769e1bbf424517003150de3f20682)
FetchContent_Declare(hashes GIT_REPOSITORY https://github.com/Kronuz/hashes.git GIT_TAG abff6fa5ac0c19c0cff5509a437700d8f77be41e)
FetchContent_Declare(termdict GIT_REPOSITORY https://github.com/Kronuz/termdict.git GIT_TAG 51278ef81c28f7f481bce23c77f5bfd964651315)
FetchContent_Declare(varint GIT_REPOSITORY https://github.com/Kronuz/varint.git GIT_TAG de3d05dbec86cd4c147b3177947f9c3b58312675)
FetchContent_Declare(residency GIT_REPOSITORY https://github.com/Kronuz/residency.git GIT_TAG adc23e8c90507e2b0f4050746969b6bf9408d0d3)
FetchContent_Declare(unicode GIT_REPOSITORY https://github.com/Kronuz/unicode.git GIT_TAG 81219c39deca91ed0186fb5eebd2afcec6f5f8c7)
FetchContent_Declare(snowball GIT_REPOSITORY https://github.com/Kronuz/snowball.git GIT_TAG 8b0c540173f8f48afc00cd8ab49ce956397992b6)
FetchContent_Declare(analysis GIT_REPOSITORY https://github.com/Kronuz/analysis.git GIT_TAG d3f94cb5cdac70c3cfa116a3fb761615c66a3f31)
# Stored records use zstd alone.
set(COMPRESSORS_DEFLATE OFF CACHE BOOL "" FORCE)
set(COMPRESSORS_LZ4 OFF CACHE BOOL "" FORCE)
FetchContent_Declare(compressors GIT_REPOSITORY https://github.com/Kronuz/compressors.git GIT_TAG 7a6cc22f8a4eaceefd098374c05cdc2935c0fd19)
FetchContent_Declare(fusefilter GIT_REPOSITORY https://github.com/Kronuz/fusefilter.git GIT_TAG b4f4ce1932c8e204fb6bebd34f83f5b10de7b51c)
# Index I/O goes through kronuz::journal::IO, from cluster's completion headers only.
FetchContent_Declare(cluster GIT_REPOSITORY https://github.com/Kronuz/cluster.git GIT_TAG ca82aada340dd86ee9deb0ddcc000c7c8501fbeb SOURCE_SUBDIR completion)
FetchContent_MakeAvailable(docset bitpack fst task hashes residency varint termdict unicode snowball analysis compressors fusefilter cluster)

add_library(strata INTERFACE)
target_include_directories(strata INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(strata INTERFACE cxx_std_20)
target_link_libraries(strata INTERFACE docset::docset bitpack::bitpack fst::fst task::task hashes::hashes residency::residency varint::varint termdict::termdict analysis::analysis compressors::compressors fusefilter::fusefilter kronuz::completion)
add_library(strata::strata ALIAS strata)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  enable_testing()
  add_executable(concepts_test test/concepts_test.cc)
  target_link_libraries(concepts_test PRIVATE strata)
  add_test(NAME concepts COMMAND concepts_test)
  add_executable(segment_test test/segment_test.cc)
  target_link_libraries(segment_test PRIVATE strata)
  target_compile_definitions(segment_test PRIVATE STRATA_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/test/fixtures")
  add_test(NAME segment COMMAND segment_test)
  add_executable(concurrency_test test/concurrency_test.cc)
  target_link_libraries(concurrency_test PRIVATE strata)
  add_test(NAME concurrency COMMAND concurrency_test)
  add_executable(files_test test/files_test.cc)
  target_link_libraries(files_test PRIVATE strata)
  add_test(NAME files COMMAND files_test)
  add_executable(lifecycle_test test/lifecycle_test.cc)
  target_link_libraries(lifecycle_test PRIVATE strata)
  add_test(NAME lifecycle COMMAND lifecycle_test)
  add_executable(index_test test/index_test.cc)
  target_link_libraries(index_test PRIVATE strata)
  add_test(NAME index COMMAND index_test)
  add_executable(faultfs_test test/faultfs_test.cc)
  target_link_libraries(faultfs_test PRIVATE strata)
  add_test(NAME faultfs COMMAND faultfs_test)
  add_executable(search_test test/search_test.cc)
  target_link_libraries(search_test PRIVATE strata)
  add_test(NAME search COMMAND search_test)
  # Fuzz targets: each target's checked-in corpus replays as a test in every build; with
  # STRATA_FUZZ (Clang) the libFuzzer binaries build too.
  option(STRATA_FUZZ "Build the libFuzzer targets (Clang)" OFF)
  foreach(target segment files)
    add_executable(fuzz_${target}_replay fuzz/fuzz_${target}.cc fuzz/replay.cc)
    target_link_libraries(fuzz_${target}_replay PRIVATE strata)
    add_test(NAME fuzz_${target}_replay COMMAND fuzz_${target}_replay "${CMAKE_CURRENT_SOURCE_DIR}/fuzz/corpus/${target}")
    if(STRATA_FUZZ)
      add_executable(fuzz_${target} fuzz/fuzz_${target}.cc)
      target_link_libraries(fuzz_${target} PRIVATE strata)
      target_compile_options(fuzz_${target} PRIVATE -fsanitize=fuzzer,address,undefined)
      target_link_options(fuzz_${target} PRIVATE -fsanitize=fuzzer,address,undefined)
    endif()
  endforeach()
  add_executable(parity tools/parity.cc)
  target_link_libraries(parity PRIVATE strata)
  add_executable(strata_engine tools/strata_engine.cc)
  target_link_libraries(strata_engine PRIVATE strata)
endif()
