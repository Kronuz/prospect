cmake_minimum_required(VERSION 3.14)
project(strata_phase0 LANGUAGES CXX)
include(FetchContent)
FetchContent_Declare(bitpack GIT_REPOSITORY https://github.com/Kronuz/bitpack.git GIT_TAG ca99608c954909f33b9a1a0ab99c357272bb773c)
FetchContent_Declare(varint GIT_REPOSITORY https://github.com/Kronuz/varint.git GIT_TAG ac6003e1ca12a2e52f6cf086d8a328d17672a558)
FetchContent_MakeAvailable(bitpack varint)
add_executable(postings postings.cc)
target_compile_features(postings PRIVATE cxx_std_20)
target_link_libraries(postings PRIVATE bitpack::bitpack varint::varint)
