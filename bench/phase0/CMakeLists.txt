cmake_minimum_required(VERSION 3.14)
project(strata_phase0 LANGUAGES CXX)
include(FetchContent)
FetchContent_Declare(bitpack GIT_REPOSITORY https://github.com/Kronuz/bitpack.git GIT_TAG 6d14df141e1dc9c8f5e709f11a55f48493744370)
FetchContent_Declare(varint GIT_REPOSITORY https://github.com/Kronuz/varint.git GIT_TAG de3d05dbec86cd4c147b3177947f9c3b58312675)
FetchContent_MakeAvailable(bitpack varint)
add_executable(postings postings.cc)
target_compile_features(postings PRIVATE cxx_std_20)
target_link_libraries(postings PRIVATE bitpack::bitpack varint::varint)
